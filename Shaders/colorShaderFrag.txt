#version 130

in vec2 fragmentPosition;
in vec4 fragmentColor;
in vec2 fragmentUV;

out vec4 color;
uniform float time;
uniform sampler2D myImage;

void main(){
    vec4 textureColor = texture(myImage, fragmentUV);

    float rawTime = mod(time*0.05, 1.0);

    vec3 color1, color2;
    float localT;

    if (rawTime < 0.25) {
        color1 = vec3(1.0, 1.0, 0.0);   // amarillo
        color2 = vec3(1.0, 0.5, 0.0);   // naranja
        localT = rawTime / 0.25;
    } else if (rawTime < 0.5) {
        color1 = vec3(1.0, 0.5, 0.0);   // naranja
        color2 = vec3(0.0, 1.0, 0.0);   // verde
        localT = (rawTime - 0.25) / 0.25;
    } else if (rawTime < 0.75) {
        color1 = vec3(0.0, 1.0, 0.0);   // verde
        color2 = vec3(0.0, 1.0, 1.0);   // aguamarina
        localT = (rawTime - 0.5) / 0.25;
    } else {
        color1 = vec3(0.0, 1.0, 1.0);   // aguamarina
        color2 = vec3(1.0, 1.0, 0.0);   // amarillo
        localT = (rawTime - 0.75) / 0.25;
    }

    vec3 finalColor = mix(color1, color2, localT);
    color = vec4(finalColor, fragmentColor.a) * textureColor;
}
